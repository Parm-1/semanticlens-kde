cmake_minimum_required(VERSION 3.20)
project(kde_embedding_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory(../vendor/llama.cpp llama-runtime)
add_executable(embedding-probe main.cpp)
target_include_directories(embedding-probe PRIVATE ../vendor/llama.cpp/vendor)
target_link_libraries(embedding-probe PRIVATE llama mtmd)
add_executable(semanticlens-film-tokenizer ../src/film-v1/tokenizer.cpp)
find_package(Qt6 REQUIRED COMPONENTS Core)
target_include_directories(semanticlens-film-tokenizer PRIVATE ../vendor/llama.cpp/vendor)
target_link_libraries(semanticlens-film-tokenizer PRIVATE llama Qt6::Core)
install(TARGETS semanticlens-film-tokenizer RUNTIME DESTINATION bin)
add_executable(sizing-check sizing_check.cpp)
target_include_directories(sizing-check PRIVATE ../vendor/llama.cpp/vendor)

# Separate film successor; embedding-probe and its source remain unchanged.
add_executable(semanticlens-film-worker ../src/film-v1/worker.cpp)
target_include_directories(semanticlens-film-worker PRIVATE ../vendor/llama.cpp/vendor)
target_link_libraries(semanticlens-film-worker PRIVATE llama)
install(TARGETS semanticlens-film-worker RUNTIME DESTINATION bin)
