cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(KDEIndexEngine LANGUAGES CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network Sql)
add_library(kde-index-core STATIC ../src/core-v5/engine.cpp)
target_include_directories(kde-index-core PUBLIC ../src/core-v5)
target_link_libraries(kde-index-core PUBLIC Qt6::Core Qt6::Sql)
add_executable(kde-index-service ../src/service-v3/main.cpp)
target_link_libraries(kde-index-service PRIVATE kde-index-core Qt6::Network)
add_executable(kde-index-cli ../src/cli/main.cpp)
target_link_libraries(kde-index-cli PRIVATE Qt6::Core Qt6::Network)
get_filename_component(KDE_INDEX_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  foreach(target kde-index-core kde-index-service kde-index-cli)
    target_compile_options(${target} PRIVATE "-ffile-prefix-map=${KDE_INDEX_SOURCE_ROOT}=kde-index")
  endforeach()
endif()
install(TARGETS kde-index-service kde-index-cli RUNTIME DESTINATION bin)
