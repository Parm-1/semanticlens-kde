cmake_minimum_required(VERSION 3.24)
project(UIRepair LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Quick Qml Network)
qt_standard_project_setup()
set(ROOT ../..)
qt_add_executable(ui-repair ${ROOT}/tools/ui_repair_harness.cpp ${ROOT}/src/ui/backend.cpp ${ROOT}/src/ui/backend.h)
target_include_directories(ui-repair PRIVATE ${ROOT}/src/ui)
target_link_libraries(ui-repair PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::Qml Qt6::Network)

qt_add_executable(ui-recovery ${ROOT}/tools/ui_repair_recovery.cpp ${ROOT}/src/ui/backend.cpp ${ROOT}/src/ui/backend.h)
target_include_directories(ui-recovery PRIVATE ${ROOT}/src/ui)
target_link_libraries(ui-recovery PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::Qml Qt6::Network)
qt_add_executable(ui-liveness ${ROOT}/tools/ui_repair_liveness.cpp ${ROOT}/src/ui/backend.cpp ${ROOT}/src/ui/backend.h)
target_include_directories(ui-liveness PRIVATE ${ROOT}/src/ui)
target_link_libraries(ui-liveness PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::Qml Qt6::Network)
