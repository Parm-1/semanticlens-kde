cmake_minimum_required(VERSION 3.22)
project(DesktopOpenV3Observer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
add_library(desktop-open-v3-observer SHARED desktop_open_v3_observer.cpp)
target_link_libraries(desktop-open-v3-observer PRIVATE Qt6::Widgets dl)
add_executable(desktop-open-v3-gio desktop_open_v3_gio.cpp)
target_link_libraries(desktop-open-v3-gio PRIVATE PkgConfig::GIO)
