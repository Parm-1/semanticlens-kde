cmake_minimum_required(VERSION 3.22)
project(FilmOpenObserver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Quick Widgets Qml Test)
add_library(film-open-observer SHARED film_open_observer.cpp)
target_link_libraries(film-open-observer PRIVATE Qt6::Quick Qt6::Widgets Qt6::Qml Qt6::Test dl)
find_package(Okular6 REQUIRED)
add_library(film-open-pdf-observer SHARED film_open_pdf_observer.cpp)
target_link_libraries(film-open-pdf-observer PRIVATE Qt6::Widgets Okular::Core dl)

add_library(film-open-konsole-observer SHARED film_open_konsole_observer.cpp)
target_link_libraries(film-open-konsole-observer PRIVATE Qt6::Widgets dl)
