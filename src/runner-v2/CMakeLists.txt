cmake_minimum_required(VERSION 3.24)
project(KDEIndexRunnerV2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(GNUInstallDirs)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network Widgets)
include(KDEInstallDirs6)
find_package(KF6Runner 6.0 REQUIRED)
find_package(KF6CoreAddons REQUIRED)
add_library(kde-index-runner-v2 MODULE runner.cpp)
target_link_libraries(kde-index-runner-v2 PRIVATE Qt6::Core Qt6::Network Qt6::Widgets KF6::Runner KF6::CoreAddons)
target_compile_definitions(kde-index-runner-v2 PRIVATE QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
install(TARGETS kde-index-runner-v2 DESTINATION ${KDE_INSTALL_PLUGINDIR}/kf6/krunner)
option(KDE_INDEX_RUNNER_V2_BUILD_SMOKE "Build local KRunner transport smoke" OFF)
if(KDE_INDEX_RUNNER_V2_BUILD_SMOKE)
  add_executable(runner-v2-smoke ../../tools/runner_v2_smoke.cpp)
  target_link_libraries(runner-v2-smoke PRIVATE Qt6::Core Qt6::Network KF6::Runner KF6::CoreAddons)
endif()
