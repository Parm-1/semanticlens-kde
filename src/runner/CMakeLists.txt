cmake_minimum_required(VERSION 3.24)
project(KDEIndexRunner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network Widgets)
find_package(KF6Runner 6.0 REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
add_library(kde-index-runner MODULE runner.cpp)
target_link_libraries(kde-index-runner PRIVATE Qt6::Core Qt6::Network Qt6::Widgets KF6::Runner KF6::CoreAddons)
target_compile_definitions(kde-index-runner PRIVATE QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
install(TARGETS kde-index-runner DESTINATION ${KDE_INSTALL_PLUGINDIR}/kf6/krunner)
option(KDE_INDEX_RUNNER_BUILD_SMOKE "Build local KRunner transport smoke" OFF)
if(KDE_INDEX_RUNNER_BUILD_SMOKE)
  add_executable(runner-smoke ../../tools/runner_smoke.cpp)
  target_link_libraries(runner-smoke PRIVATE Qt6::Core Qt6::Network KF6::Runner KF6::CoreAddons)
endif()
