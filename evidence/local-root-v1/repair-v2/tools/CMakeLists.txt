cmake_minimum_required(VERSION 3.24)
project(LocalRootFrozenCandidate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Network Sql)
add_library(core STATIC ../source/engine.cpp)
target_include_directories(core PUBLIC ../source)
target_link_libraries(core PUBLIC Qt6::Core Qt6::Sql)
add_executable(local-root-service ../source/service.cpp)
target_link_libraries(local-root-service PRIVATE core Qt6::Network)
add_executable(local-root-controls local_root_controls.cpp)
target_link_libraries(local-root-controls PRIVATE core)
