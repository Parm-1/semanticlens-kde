cmake_minimum_required(VERSION 3.24)
project(KDEIndex VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(KDE_INDEX_BUILD_ENGINE "Build the local search service and CLI" ON)
option(KDE_INDEX_BUILD_RUNNER "Build the native KRunner adapter" ON)
option(KDE_INDEX_BUILD_PROBE "Build the pinned CPU embedding producer" ON)

find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickControls2 Network)
find_package(KF6Kirigami REQUIRED)
find_package(KF6IconThemes REQUIRED)
qt_standard_project_setup(REQUIRES 6.5)

if(KDE_INDEX_BUILD_PROBE)
    add_subdirectory(probe)
    install(TARGETS embedding-probe RUNTIME DESTINATION bin)
endif()
if(KDE_INDEX_BUILD_ENGINE)
    add_subdirectory(engine)
endif()
if(KDE_INDEX_BUILD_RUNNER)
    add_subdirectory(src/runner-v4)
endif()

qt_add_executable(kde-index src/ui/main.cpp src/ui/backend.cpp src/ui/backend.h)
target_link_libraries(kde-index PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Network KF6::IconThemes)
set_source_files_properties(src/ui/Main.qml PROPERTIES QT_RESOURCE_ALIAS Main.qml)
qt_add_qml_module(kde-index URI org.kde.kdeindex VERSION 1.0 QML_FILES src/ui/Main.qml)
install(TARGETS kde-index RUNTIME DESTINATION bin)
install(PROGRAMS tools/model_download.py DESTINATION share/kde-index)
install(FILES packaging/org.kdeindex.Search.desktop DESTINATION share/applications)
install(FILES packaging/org.kdeindex.Search.metainfo.xml DESTINATION share/metainfo)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION share/licenses/kde-index)
install(DIRECTORY LICENSES/ DESTINATION share/licenses/kde-index)

option(KDE_INDEX_BUILD_UI_SMOKE "Build the actual service/window validation runner" OFF)
if(KDE_INDEX_BUILD_UI_SMOKE)
    qt_add_executable(ui-smoke tools/ui_smoke.cpp src/ui/backend.cpp src/ui/backend.h)
    target_include_directories(ui-smoke PRIVATE src/ui)
    target_link_libraries(ui-smoke PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Network KF6::IconThemes)
    qt_add_qml_module(ui-smoke URI org.kde.kdeindex VERSION 1.0 OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/ui-smoke-qml" QML_FILES src/ui/Main.qml)
endif()
