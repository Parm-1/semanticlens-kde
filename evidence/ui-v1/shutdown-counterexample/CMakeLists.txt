cmake_minimum_required(VERSION 3.24)
project(UILifetime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network)
foreach(candidate before fixed)
  set(backend "${CMAKE_CURRENT_SOURCE_DIR}/backend-before.cpp")
  if(candidate STREQUAL "fixed")
    set(backend "/home/parm/Documents/KDE Index/src/ui/backend.cpp")
  endif()
  add_executable(lifetime-${candidate} "/home/parm/Documents/KDE Index/tools/ui_lifetime.cpp" "${backend}" "/home/parm/Documents/KDE Index/src/ui/backend.h")
  target_include_directories(lifetime-${candidate} PRIVATE "/home/parm/Documents/KDE Index/src/ui")
  target_link_libraries(lifetime-${candidate} PRIVATE Qt6::Core Qt6::Gui Qt6::Network)
  target_compile_options(lifetime-${candidate} PRIVATE -fsanitize=address -fno-omit-frame-pointer -g)
  target_link_options(lifetime-${candidate} PRIVATE -fsanitize=address)
endforeach()

find_package(Qt6 REQUIRED COMPONENTS Widgets Qml Quick QuickControls2)
find_package(KF6IconThemes REQUIRED)
qt_standard_project_setup(REQUIRES 6.5)
set_source_files_properties("/home/parm/Documents/KDE Index/src/ui/Main.qml" PROPERTIES QT_RESOURCE_ALIAS Main.qml)
qt_add_executable(gui-before "/home/parm/Documents/KDE Index/src/ui/main.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/backend-before.cpp" "/home/parm/Documents/KDE Index/src/ui/backend.h")
target_include_directories(gui-before PRIVATE "/home/parm/Documents/KDE Index/src/ui")
target_link_libraries(gui-before PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Qml Qt6::Quick Qt6::QuickControls2 KF6::IconThemes)
qt_add_qml_module(gui-before URI org.kde.kdeindex VERSION 1.0 QML_FILES "/home/parm/Documents/KDE Index/src/ui/Main.qml")
