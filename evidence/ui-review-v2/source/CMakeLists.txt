cmake_minimum_required(VERSION 3.24)
project(UIOwnershipReview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network)
qt_standard_project_setup()
if(NOT UI_CANDIDATE)
 message(FATAL_ERROR "UI_CANDIDATE must be the frozen UI-v2 candidate path")
endif()
qt_add_executable(ownership ownership.cpp ${UI_CANDIDATE}/src/ui/backend.cpp ${UI_CANDIDATE}/src/ui/backend.h)
target_include_directories(ownership PRIVATE ${UI_CANDIDATE}/src/ui)
target_link_libraries(ownership PRIVATE Qt6::Core Qt6::Gui Qt6::Network)
qt_add_executable(phase phase.cpp ${UI_CANDIDATE}/src/ui/backend.cpp ${UI_CANDIDATE}/src/ui/backend.h)
target_include_directories(phase PRIVATE ${UI_CANDIDATE}/src/ui)
target_link_libraries(phase PRIVATE Qt6::Core Qt6::Gui Qt6::Network)
