cmake_minimum_required(VERSION 3.24)
project(ServiceWatchV1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Network Sql)
add_library(core STATIC ../../../src/core/engine.cpp)
target_include_directories(core PUBLIC ../../../src/core)
target_link_libraries(core PUBLIC Qt6::Core Qt6::Sql)
add_executable(service-original ../../../src/service/main.cpp)
target_link_libraries(service-original PRIVATE core Qt6::Network)
add_executable(service-v2 src/service-v2/main.cpp)
target_link_libraries(service-v2 PRIVATE core Qt6::Network)
