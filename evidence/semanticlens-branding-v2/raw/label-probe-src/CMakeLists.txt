cmake_minimum_required(VERSION 3.24)
project(SemanticLensLabelProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickControls2 Network)
find_package(KF6Kirigami REQUIRED)
find_package(KF6IconThemes REQUIRED)
qt_standard_project_setup(REQUIRES 6.5)
set(SOURCE_ROOT /public-clean-v2/source/kde-index-source)
qt_add_executable(ui-label main.cpp ${SOURCE_ROOT}/src/ui-v4/backend.cpp ${SOURCE_ROOT}/src/ui-v4/backend.h)
target_include_directories(ui-label PRIVATE ${SOURCE_ROOT}/src/ui-v4)
target_link_libraries(ui-label PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Network KF6::IconThemes)
set_source_files_properties(${SOURCE_ROOT}/src/ui-v4/Main.qml PROPERTIES QT_RESOURCE_ALIAS Main.qml)
qt_add_qml_module(ui-label URI org.kde.kdeindex VERSION 1.0 QML_FILES ${SOURCE_ROOT}/src/ui-v4/Main.qml)
